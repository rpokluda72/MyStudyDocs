1. Introduction & Connecting

PostgreSQL (Postgres) is an open-source relational database. Data is stored in tables with rows and columns, and SQL (Structured Query Language) is used to create, read, update, and delete that data.

psql — Built-in CLI Client

psql ships with every PostgreSQL installation. Use it to connect and run SQL interactively.

# connect as default postgres superuser

psql -U postgres

# connect to a specific database

psql -U postgres -d mydb

# explicit host and port

psql -U postgres -h localhost -p 5432

psql Meta-Commands

Meta-commands start with \ and are psql-specific shortcuts, not SQL.

2. Managing Databases

Create, Switch, Drop

-- create a new database

CREATE DATABASE shop;

-- connect to it (psql meta-command)

\c shop

-- drop database (must not be connected to it)

DROP DATABASE shop;

DROP DATABASE IF EXISTS shop; -- safe version

Create User & Grant Access

CREATE USER myuser WITH PASSWORD 'secret';

GRANT ALL PRIVILEGES ON DATABASE shop TO myuser;

Connection String Format

Used by applications, ORMs, and GUI tools to connect:

postgresql://username:password@localhost:5432/dbname

CREATE TABLE Example

CREATE TABLE products (

id SERIAL PRIMARY KEY,

name VARCHAR(100) NOT NULL,

price NUMERIC(10, 2) NOT NULL,

stock INTEGER DEFAULT 0,

created_at TIMESTAMP DEFAULT NOW()

);

Alter Table

-- add a column

ALTER TABLE products ADD COLUMN category VARCHAR(50);

-- rename a column

ALTER TABLE products RENAME COLUMN stock TO quantity;

-- drop a column

ALTER TABLE products DROP COLUMN category;

4. INSERT — Adding Rows

Single Row

INSERT INTO products (name, price, stock)

VALUES ('Laptop', 999.99, 10);

Multiple Rows at Once

INSERT INTO products (name, price, stock) VALUES

('Mouse', 29.99, 50),

('Keyboard', 79.99, 30),

('Monitor', 349.00, 5);

INSERT … RETURNING

Returns the inserted row(s) — handy for getting the auto-generated id.

INSERT INTO products (name, price)

VALUES ('Webcam', 89.99)

RETURNING id, name;

5. SELECT — Querying Data

Basic SELECT

-- all columns, all rows

SELECT * FROM products;

-- specific columns

SELECT name, price FROM products;

-- column alias

SELECT name, price * 1.2 AS price_with_tax FROM products;

WHERE — Filtering Rows

SELECT * FROM products WHERE price < 100;

SELECT * FROM products WHERE stock > 0 AND price < 200;

SELECT * FROM products WHERE name ILIKE '%key%';

SELECT * FROM products WHERE id IN (1, 2, 3);

ORDER BY, LIMIT, OFFSET

-- descending order

SELECT * FROM products ORDER BY price DESC;

-- top 5 cheapest

SELECT * FROM products ORDER BY price ASC LIMIT 5;

-- pagination: page 3, 10 items per page

SELECT * FROM products ORDER BY id LIMIT 10 OFFSET 20;

Aggregate Functions

SELECT

COUNT(*) AS total_products,

AVG(price) AS avg_price,

SUM(stock) AS total_stock,

MIN(price) AS cheapest,

MAX(price) AS most_expensive

FROM products;

GROUP BY & HAVING

-- count orders per product

SELECT product_id, COUNT(*) AS order_count

FROM orders

GROUP BY product_id

HAVING COUNT(*) > 1

ORDER BY order_count DESC;

Note: HAVING filters on aggregate results; WHERE filters on individual rows before aggregation.

6. UPDATE — Modifying Rows

Basic UPDATE

-- update one field

UPDATE products

SET price = 899.99

WHERE name = 'Laptop';

-- update multiple fields

UPDATE products

SET price = 25.00, stock = 60

WHERE name = 'Mouse';

UPDATE … RETURNING

UPDATE products

SET stock = stock - 1

WHERE id = 1

RETURNING id, name, stock;

Note: Always include WHERE — without it every row in the table is updated.

7. DELETE — Removing Rows

-- delete one row

DELETE FROM products WHERE name = 'Monitor';

-- delete all rows (keeps table structure)

DELETE FROM products;

-- faster bulk delete, resets SERIAL counter

TRUNCATE TABLE products;

-- drop entire table

DROP TABLE products;

DROP TABLE IF EXISTS products; -- no error if missing

Note: Always include WHERE — without it every row is deleted.

DELETE … RETURNING

DELETE FROM products

WHERE stock = 0

RETURNING id, name;

9. JOINs — Combining Tables

JOINs combine rows from two or more tables based on a matching column (usually a foreign key).

Setup — Orders Table

CREATE TABLE orders (

id SERIAL PRIMARY KEY,

product_id INTEGER REFERENCES products(id),

quantity INTEGER NOT NULL,

ordered_at TIMESTAMP DEFAULT NOW()

);

INSERT INTO orders (product_id, quantity) VALUES

(1, 2),

(2, 5);

JOIN Types

Type

Returns

INNER JOIN

Only rows with a match in both tables

LEFT JOIN

All rows from the left table + matching rows from right (NULL if no match)

RIGHT JOIN

All rows from the right table + matching rows from left (NULL if no match)

FULL OUTER JOIN

All rows from both tables — NULL where there is no match

CROSS JOIN

Every combination of rows (cartesian product) — no ON clause needed

INNER JOIN

Returns only rows that have a match in both tables.

SELECT o.id, p.name, p.price, o.quantity,

(p.price * o.quantity) AS total

FROM orders o

JOIN products p ON o.product_id = p.id;

-- JOIN without qualifier = INNER JOIN

LEFT JOIN Example

-- show ALL products, even those with no orders

SELECT p.name, p.price, o.quantity

FROM products p

LEFT JOIN orders o ON p.id = o.product_id;

-- products with no orders will have NULL in quantity column

RIGHT JOIN Example

Returns ALL rows from the RIGHT table (orders), plus matched rows from the left (products). Where there is no matching product, the product columns are NULL.

-- show ALL orders, even if the linked product was deleted

SELECT p.name, p.price, o.id AS order_id, o.quantity

FROM products p

RIGHT JOIN orders o ON p.id = o.product_id;

-- orders with no matching product → NULL in p.name, p.price

-- equivalent written the other way (LEFT JOIN with tables swapped):

SELECT p.name, p.price, o.id AS order_id, o.quantity

FROM orders o

LEFT JOIN products p ON o.product_id = p.id;

FULL OUTER JOIN Example

Returns ALL rows from BOTH tables. Where there is no match, the missing side is filled with NULL. Combines LEFT JOIN + RIGHT JOIN results.

-- show all products AND all orders, matched where possible

SELECT p.name, o.id AS order_id, o.quantity

FROM products p

FULL OUTER JOIN orders o ON p.id = o.product_id;

-- products with no orders → NULL in order_id, quantity

-- orders with no product → NULL in name

-- find unmatched rows on BOTH sides

SELECT p.name, o.id AS order_id

FROM products p

FULL OUTER JOIN orders o ON p.id = o.product_id

WHERE p.id IS NULL OR o.id IS NULL;

CROSS JOIN Example

Returns every combination of rows from both tables (cartesian product). No ON clause needed. Result rows = left_count × right_count.

-- every product paired with every available size

-- 4 products × 3 sizes = 12 result rows

SELECT p.name, s.label AS size, p.price

FROM products p

CROSS JOIN sizes s;

-- practical use: generate a full discount price matrix

SELECT p.name,

d.percent AS discount_pct,

ROUND(p.price * (1 - d.percent / 100.0), 2) AS final_price

FROM products p

CROSS JOIN discounts d

ORDER BY p.name, d.percent;

10. Quick Reference

SQL Statement — Clause Order

SELECT columns -- what to return

FROM table -- source table

JOIN other_table ON condition -- combine tables

WHERE filter_condition -- filter rows

GROUP BY column -- group for aggregates

HAVING aggregate_condition -- filter groups

ORDER BY column ASC | DESC -- sort result

LIMIT 10 OFFSET 20; -- paginate

Note: Execution order: FROM → JOIN → WHERE → GROUP BY → HAVING → SELECT → ORDER BY → LIMIT

Useful Built-in Functions

-- String

UPPER(name) LOWER(name) LENGTH(name)

TRIM(name) SUBSTRING(name, 1, 3)

CONCAT(first, ' ', last) name || ' ' || last -- || is concat

-- Numeric

ROUND(price, 2) CEIL(price) FLOOR(price) ABS(val)

-- Date / Time

NOW() -- current timestamp

CURRENT_DATE -- today's date only

DATE_PART('year', created_at) -- extract year

AGE(created_at) -- interval since that date

-- Type casting

price::TEXT CAST(price AS TEXT)

'2024-01-15'::DATE -- cast string to date
