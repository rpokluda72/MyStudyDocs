1. Co je .NET?

.NET je bezplatná, multiplatformní a open source vývojářská platforma vytvořená společností Microsoft. Umožňuje vyvíjet aplikace v jazycích C#, F# a Visual Basic pro Windows, Linux i macOS.
Zkratka: .NET je nástupce .NET Core (od verze 5) a moderní náhrada za starší .NET Framework (pouze Windows). Dnes se říká jednoduše „.NET".

7. Správa paměti a Garbage Collector

7.1 Jak funguje správa paměti v .NET

.NET automaticky spravuje paměť pomocí Garbage Collectoru (GC). Vývojář nemusí manuálně uvolňovat paměť jako v C++.

7.3 IDisposable a using

GC spravuje paměť, ale ne jiné zdroje (soubory, síťová připojení, databáze). Pro tyto zdroje slouží rozhraní IDisposable a příkaz using:

Příklad – IDisposable a using
// using – zajistí automatické uvolnění zdroje (Dispose)
using (FileStream fs = new FileStream("data.txt", FileMode.Open))
{
    // práce se souborem
} // Dispose() se zavolá automaticky i při výjimce
 
// Modernější zápis (using declaration)
using var fs = new FileStream("data.txt", FileMode.Open);
// Dispose se zavolá na konci bloku
 
// IDisposable implementace vlastní třídy
class MojeTrida : IDisposable
{
    private bool _disposed = false;
 
    public void Dispose()
    {
        if (!_disposed)
        {
            // uvolnění zdrojů
            _disposed = true;
        }
    }
}
Pravidlo: Vždy použijte using pro objekty implementující IDisposable (FileStream, HttpClient, DbConnection, SqlCommand, ...). Zabraňuje únikům zdrojů (resource leaks).

Příklad – Minimal API
// Program.cs – ASP.NET Core Minimal API
var builder = WebApplication.CreateBuilder(args);
 
// Registrace služeb (Dependency Injection)
builder.Services.AddSingleton<IWeatherService, WeatherService>();
 
var app = builder.Build();
 
// Definice endpointů
app.MapGet("/", () => "Ahoj z .NET!");
 
app.MapGet("/pocasi", (IWeatherService service) =>
{
    var data = service.GetForecast();
    return Results.Ok(data);
});
 
app.MapPost("/pocasi", (WeatherData data) =>
{
    // zpracování POST requestu
    return Results.Created($"/pocasi/{data.Id}", data);

app.Run();

9.2 MVC Controller API

Příklad – Controller
// Controllers/PocasiController.cs
using Microsoft.AspNetCore.Mvc;
 
[ApiController]
[Route("api/[controller]")]
public class PocasiController : ControllerBase
{
    private readonly IWeatherService _service;
    private readonly ILogger<PocasiController> _logger;
 
    public PocasiController(IWeatherService service,
                             ILogger<PocasiController> logger)
    {
        _service = service;
        _logger = logger;
    }
 
    // GET api/pocasi
    [HttpGet]
    public ActionResult<IEnumerable<WeatherForecast>> Get()
    {
        _logger.LogInformation("Načítám předpověď počasí");
        return Ok(_service.GetForecast());
    }
 
    // GET api/pocasi/5
    [HttpGet("{id}")]
    public ActionResult<WeatherForecast> GetById(int id)
    {
        var item = _service.GetById(id);
        return item is null ? NotFound() : Ok(item);
    }
 
    // POST api/pocasi
    [HttpPost]
    public ActionResult<WeatherForecast> Create(WeatherForecast forecast)
    {
        _service.Add(forecast);
        return CreatedAtAction(nameof(GetById), new { id = forecast.Id }, forecast);
    }
}

services.AddScoped<IObjednavkaService, ObjednavkaService>();

    options.UseSqlServer(connString)); // DbContext je vždy Scoped
 
// Transient – nová instance na každé vyžádání
services.AddTransient<IEmailService, EmailService>();
 
// Rozhraní – bez interface, registrace konkrétní třídy
services.AddSingleton<LogovaniService>();
 
var app = builder.Build();
 
// Controller nebo Minimal API automaticky dostane injektované závislosti
app.MapGet("/objednavky", (IObjednavkaService svc) =>
    svc.GetAll());

public class Kalkulacka

        return a / b;
    }
}

{

    [Fact]  // jednoduchý test bez parametrů
    public void Secti_DveKladnaCisla_VratisokretSoucet()
    {
        // Arrange (připrav)
        int a = 3, b = 4;
 
        // Act (proveď)
        int vysledek = _kalkulacka.Secti(a, b);
 
        // Assert (ověř)
        Assert.Equal(7, vysledek);
    }
 
    [Theory]  // parametrizovaný test
    [InlineData(10, 2, 5)]
    [InlineData(9, 3, 3)]
    [InlineData(0, 5, 0)]
    public void Vydel_PlatneVstupy_VratiSpravnyVysledek(
        int a, int b, int ocekavany)
    {
        int vysledek = _kalkulacka.Vydel(a, b);
        Assert.Equal(ocekavany, vysledek);
    }
 
    [Fact]
    public void Vydel_NuloutVmenovateli_HodivyVyjimku()
    {
        Assert.Throws<DivideByZeroException>(
            () => _kalkulacka.Vydel(5, 0));

using Moq;

public interface IEmailService

    {
