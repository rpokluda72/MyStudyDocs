1. SELECT Fundamentals

SELECT retrieves data from one or more tables. It is the most used SQL statement — mastering it is the foundation of everything else.

SELECT what_to_return ← columns, expressions, *

FROM source_table ← where data lives

WHERE filter_condition ← row filter (optional)

ORDER BY column ASC|DESC ← sort (optional)

LIMIT n; ← row cap (optional)

All Columns vs Specific Columns

-- all columns (avoid in production — expensive, fragile)

SELECT * FROM employees;

-- specific columns (preferred)

SELECT first_name, last_name, salary FROM employees;

Column Aliases — AS

SELECT

first_name AS first,

salary * 12 AS annual_salary,

UPPER(last_name) AS last_name_upper,

salary * 0.2 AS tax_estimate

FROM employees;

DISTINCT — Remove Duplicates

-- unique departments

SELECT DISTINCT department FROM employees;

-- unique combination of department + job_title

SELECT DISTINCT department, job_title FROM employees;

Note: DISTINCT applies to the full row, not just the first column.

Expressions & CASE WHEN

SELECT

name,

salary,

CASE

WHEN salary >= 90000 THEN 'Senior'

WHEN salary >= 60000 THEN 'Mid'

ELSE 'Junior'

END AS level,

CASE WHEN bonus IS NULL THEN 0 ELSE bonus END AS bonus_safe

FROM employees;

Logical Execution Order

SQL clauses execute in a different order than they are written:

Written order: FROM → WHERE → GROUP BY → HAVING → SELECT → ORDER BY → LIMIT

Key implication: You CANNOT use a SELECT alias in WHERE or GROUP BY

because WHERE runs before SELECT is evaluated.

✗ SELECT salary * 12 AS annual FROM emp WHERE annual > 100000 -- error

✓ SELECT salary * 12 AS annual FROM emp WHERE salary * 12 > 100000 -- correct

2. Filtering & Sorting

WHERE — Comparison Operators

SELECT * FROM employees WHERE salary > 70000;

SELECT * FROM employees WHERE department <> 'HR'; -- != also works

SELECT * FROM employees WHERE hire_date >= '2022-01-01';

AND, OR, NOT

SELECT * FROM employees

WHERE salary > 60000 AND department = 'Engineering';

SELECT * FROM employees

WHERE department = 'HR' OR department = 'Finance';

SELECT * FROM employees

WHERE NOT department = 'HR';

-- use parentheses to control precedence (AND binds tighter than OR)

SELECT * FROM employees

WHERE (department = 'HR' OR department = 'Finance') AND salary > 50000;

BETWEEN, IN, LIKE, IS NULL

-- BETWEEN (inclusive on both ends)

SELECT * FROM employees WHERE salary BETWEEN 50000 AND 90000;

-- IN (matches any value in the list)

SELECT * FROM employees WHERE department IN ('HR', 'Finance', 'Legal');

-- LIKE pattern matching (% = any chars, _ = one char)

SELECT * FROM employees WHERE last_name LIKE 'Sm%'; -- starts with Sm

SELECT * FROM employees WHERE last_name LIKE '_mith'; -- 5-char, ends ith

SELECT * FROM employees WHERE email LIKE '%@gmail.com';

-- NULL checks (always use IS NULL, never = NULL)

SELECT * FROM employees WHERE manager_id IS NULL; -- no manager = top level

SELECT * FROM employees WHERE bonus IS NOT NULL;

Note: NULL = NULL evaluates to NULL (unknown), not TRUE. Always use IS NULL / IS NOT NULL.

ORDER BY

-- single column

SELECT * FROM employees ORDER BY salary DESC;

-- multiple columns — sort by dept asc, then salary desc within each dept

SELECT * FROM employees ORDER BY department ASC, salary DESC;

-- NULLs sort last by default in most DBs; explicit control:

SELECT * FROM employees ORDER BY bonus DESC NULLS LAST; -- PostgreSQL

LIMIT / OFFSET — Pagination

-- top 5 highest paid

SELECT * FROM employees ORDER BY salary DESC LIMIT 5;

-- page 3, 10 rows per page (skip first 20)

SELECT * FROM employees ORDER BY id LIMIT 10 OFFSET 20;

-- SQL Server / MS Access equivalent:

SELECT TOP 5 * FROM employees ORDER BY salary DESC;

SUM(salary) AS payroll,

GROUP BY — Aggregate Per Group

-- headcount and average salary per department

SELECT

COUNT(*) AS headcount,

AVG(salary) AS avg_salary,

FROM employees

-- group by multiple columns

SELECT department, job_title, COUNT(*) AS cnt

FROM employees

GROUP BY department, job_title;

Note: Every column in SELECT must either be in GROUP BY or be inside an aggregate function.

4. JOINs

JOINs combine rows from two or more tables using a related column. The most common pattern is linking a foreign key to a primary key.

employees table departments table

┌────┬──────────┬────────┐ ┌────┬─────────────┐

│ id │ name │dept_id │ │ id │ name │

├────┼──────────┼────────┤ ├────┼─────────────┤

│ 1 │ Alice │ 1 │ │ 1 │ Engineering │

│ 2 │ Bob │ 2 │ │ 2 │ HR │

│ 3 │ Carol │ 1 │ │ 3 │ Finance │

│ 4 │ Dave │ NULL │ └────┴─────────────┘

└────┴──────────┴────────┘

INNER JOIN → Alice, Bob, Carol (Dave excluded — no dept)

LEFT JOIN → Alice, Bob, Carol, Dave (Dave dept = NULL)

RIGHT JOIN → Alice, Bob, Carol + Finance row with NULL employee

FULL OUTER → all of the above combined

INNER JOIN

-- only employees that have a matching department

SELECT e.name, e.salary, d.name AS department

FROM employees e

INNER JOIN departments d ON e.dept_id = d.id;

-- INNER JOIN = JOIN (INNER is optional)

LEFT JOIN

-- all employees, even those with no department

SELECT e.name, d.name AS department

FROM employees e

LEFT JOIN departments d ON e.dept_id = d.id;

-- Dave appears with department = NULL

-- find employees with NO department (anti-join pattern)

SELECT e.name

FROM employees e

LEFT JOIN departments d ON e.dept_id = d.id

WHERE d.id IS NULL;

RIGHT JOIN

-- all departments, even those with no employees

SELECT e.name, d.name AS department

FROM employees e

RIGHT JOIN departments d ON e.dept_id = d.id;

-- Finance row appears with e.name = NULL

-- Tip: RIGHT JOIN can always be rewritten as LEFT JOIN (swap tables)

FULL OUTER JOIN

-- all employees AND all departments

SELECT e.name, d.name AS department

FROM employees e

FULL OUTER JOIN departments d ON e.dept_id = d.id;

-- Dave (no dept) and Finance (no employee) both appear with NULLs

CROSS JOIN

-- every employee paired with every department (cartesian product)

-- result rows = employees_count × departments_count

SELECT e.name, d.name AS department

FROM employees e

CROSS JOIN departments d;

-- practical: generate all size × color combinations for a product

SELECT s.label AS size, c.name AS color

FROM sizes s CROSS JOIN colors c;

Self JOIN

-- employees with their manager name (manager is also in employees table)

SELECT

e.name AS employee,

m.name AS manager

FROM employees e

LEFT JOIN employees m ON e.manager_id = m.id;

-- LEFT JOIN ensures top-level employees (no manager) appear with NULL

JOIN on Multiple Conditions

SELECT o.id, o.amount, d.rate

FROM orders o

JOIN discount_rules d

ON o.customer_id = d.customer_id

AND o.amount BETWEEN d.min_amount AND d.max_amount;

JOIN Type

Returns

INNER JOIN

Rows that match in BOTH tables

LEFT JOIN

All left rows + matched right (NULL if no match)

RIGHT JOIN

All right rows + matched left (NULL if no match)

FULL OUTER JOIN

All rows from both tables (NULL where no match)

CROSS JOIN

Cartesian product — every left row × every right row

Self JOIN

Table joined with itself using aliases

5. Subqueries & CTEs

A subquery is a SELECT nested inside another SQL statement. A CTE (Common Table Expression) is a named, reusable subquery defined with WITH.

Scalar Subquery — Single Value

-- employees earning above the company average

SELECT name, salary

FROM employees

WHERE salary > (SELECT AVG(salary) FROM employees),

ORDER BY salary DESC;

-- use scalar subquery in SELECT column

SELECT name, salary,

salary - (SELECT AVG(salary) FROM employees) AS diff_from_avg

FROM employees;

IN Subquery

-- employees in departments located in New York

SELECT name FROM employees

WHERE dept_id IN (

SELECT id FROM departments WHERE city = 'New York'

);

EXISTS vs IN

-- EXISTS: true if subquery returns at least one row

SELECT name FROM customers c

WHERE EXISTS (

SELECT 1 FROM orders o WHERE o.customer_id = c.id

);

-- NOT EXISTS: customers who never placed an order (anti-join)

SELECT name FROM customers c

WHERE NOT EXISTS (

SELECT 1 FROM orders o WHERE o.customer_id = c.id

);

Correlated Subquery

A correlated subquery references columns from the outer query — it runs once per outer row.

-- employees who earn more than the average of their own department

SELECT e1.name, e1.salary, e1.department

FROM employees e1

WHERE e1.salary > (

SELECT AVG(e2.salary)

FROM employees e2

WHERE e2.department = e1.department ← references outer query

);

Note: Correlated subqueries can be slow on large tables — a window function (AVG OVER PARTITION BY) is usually faster.

Subquery in FROM — Derived Table

-- department-level stats, then filter on those stats

SELECT dept_stats.department, dept_stats.avg_salary

FROM (

SELECT department, AVG(salary) AS avg_salary

FROM employees

GROUP BY department

) AS dept_stats

WHERE dept_stats.avg_salary > 75000;

CTE — WITH Clause

A CTE gives the subquery a name and makes the query easier to read. It can be referenced multiple times in the same query.

-- single CTE

WITH high_earners AS (

SELECT * FROM employees WHERE salary > 80000

)

SELECT department, COUNT(*) AS cnt

FROM high_earners

GROUP BY department;

-- multiple CTEs chained together

WITH

dept_avg AS (

SELECT department, AVG(salary) AS avg_sal

FROM employees GROUP BY department

),

top_depts AS (

SELECT * FROM dept_avg WHERE avg_sal > 70000

)

SELECT e.name, e.salary, t.avg_sal

FROM employees e

JOIN top_depts t ON e.department = t.department;

6. Window Functions

Window functions compute a value across a set of rows related to the current row — without collapsing them into one row (unlike GROUP BY). They use the OVER() clause.

SELECT

name,

salary,

AVG(salary) OVER (PARTITION BY department) AS dept_avg

──── ────────────────────────

│ window definition

│

function name PARTITION BY = group rows (like GROUP BY)

ORDER BY = defines row order within window

ROWS/RANGE = frame size (optional)

Result: each row keeps its own data PLUS the window aggregate value.

OVER Clause Anatomy

FUNCTION() OVER (

PARTITION BY column -- reset window per group

ORDER BY column ASC|DESC -- row order within window

ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW -- frame

)

Ranking Functions

SELECT

name,

department,

salary,

ROW_NUMBER() OVER (PARTITION BY department ORDER BY salary DESC) AS row_num,

RANK() OVER (PARTITION BY department ORDER BY salary DESC) AS rank,

DENSE_RANK() OVER (PARTITION BY department ORDER BY salary DESC) AS dense_rank

FROM employees;

Top N Per Group — Common Interview Pattern

-- top 2 earners per department

WITH ranked AS (

SELECT name, department, salary,

ROW_NUMBER() OVER (PARTITION BY department ORDER BY salary DESC) AS rn

FROM employees

)

SELECT name, department, salary

FROM ranked

WHERE rn <= 2;

Running Total — SUM OVER

SELECT

order_date,

amount,

SUM(amount) OVER (ORDER BY order_date) AS running_total,

SUM(amount) OVER (PARTITION BY customer_id ORDER BY order_date) AS customer_running_total

FROM orders;

LAG & LEAD — Access Adjacent Rows

SELECT

order_date,

amount,

LAG(amount) OVER (ORDER BY order_date) AS prev_amount,

LEAD(amount) OVER (ORDER BY order_date) AS next_amount,

amount - LAG(amount, 1, 0) OVER (ORDER BY order_date) AS change

FROM orders;

-- LAG(col, offset, default) — offset rows back, default if no prior row

7. Data Modification

INSERT

-- single row

INSERT INTO employees (name, department, salary)

VALUES ('Alice', 'Engineering', 85000);

-- multiple rows

INSERT INTO employees (name, department, salary) VALUES

('Bob', 'HR', 62000),

('Carol', 'Finance', 71000),

('Dave', 'Engineering', 90000);

-- INSERT from SELECT (copy rows)

INSERT INTO employees_archive

SELECT * FROM employees WHERE hire_date < '2020-01-01';

UPDATE

-- update one column

UPDATE employees

SET salary = 95000

WHERE name = 'Alice';

-- update multiple columns

UPDATE employees

SET salary = salary * 1.10, department = 'Senior Engineering'

WHERE department = 'Engineering' AND salary < 80000;

-- update using a subquery

UPDATE employees

SET department = 'Leadership'

WHERE id IN (SELECT manager_id FROM employees WHERE manager_id IS NOT NULL);

Note: Always use WHERE in UPDATE. Without it, every row is updated.

DELETE

-- delete specific rows

DELETE FROM employees WHERE department = 'HR';

-- delete using a subquery

DELETE FROM orders

WHERE customer_id IN (

SELECT id FROM customers WHERE active = FALSE

);

-- delete all rows but keep table structure

DELETE FROM employees;

-- faster bulk delete (cannot be rolled back in most DBs)

TRUNCATE TABLE employees;

Indexes

Indexes speed up SELECT queries on large tables. They come at a cost: slower INSERT/UPDATE/DELETE and extra disk space.

-- single column index

CREATE INDEX idx_emp_dept ON employees (department);

-- composite index (order matters — most selective column first)

CREATE INDEX idx_emp_dept_salary ON employees (department, salary);

-- unique index (also enforces uniqueness)

CREATE UNIQUE INDEX idx_emp_email ON employees (email);

-- drop index

DROP INDEX idx_emp_dept;

ALTER TABLE

-- add a column

ALTER TABLE employees ADD COLUMN phone VARCHAR(20);

-- drop a column

ALTER TABLE employees DROP COLUMN phone;

-- rename a column (PostgreSQL / MySQL 8+)

ALTER TABLE employees RENAME COLUMN phone TO mobile;

-- add a foreign key constraint

ALTER TABLE employees

ADD CONSTRAINT fk_dept FOREIGN KEY (dept_id) REFERENCES departments(id);

-- drop a constraint

ALTER TABLE employees DROP CONSTRAINT fk_dept;

DROP TABLE

DROP TABLE employees;

DROP TABLE IF EXISTS employees; -- no error if table does not exist

DROP TABLE departments CASCADE; -- also drops dependent objects (FK refs)

9. Transactions & ACID

A transaction is a group of SQL statements that execute as a single unit — either all succeed or all fail.

ACID Properties

Property

Meaning

Example

Atomicity

All operations succeed or none do

Transfer money: debit + credit both happen or neither does

Consistency

DB moves from one valid state to another

Account balance cannot go negative if CHECK constraint exists

Isolation

Concurrent transactions do not interfere

Two transfers at the same time see consistent balances

Durability

Committed data survives crashes

After COMMIT, data is written to disk even if server restarts

Basic Transaction

BEGIN; -- start transaction

UPDATE accounts SET balance = balance - 500 WHERE id = 1;

UPDATE accounts SET balance = balance + 500 WHERE id = 2;

COMMIT; -- save both changes permanently

-- if something goes wrong:

ROLLBACK; -- undo all changes since BEGIN

SAVEPOINT — Partial Rollback

BEGIN;

INSERT INTO orders (customer_id, amount) VALUES (1, 200);

SAVEPOINT after_order;

INSERT INTO payments (order_id, amount) VALUES (1, 200);

-- payment failed — roll back only the payment, keep the order

ROLLBACK TO after_order;

COMMIT; -- commits only the order insert

10. Interview Patterns Cheat Sheet

Find Duplicates

-- find emails that appear more than once

SELECT email, COUNT(*) AS cnt

FROM employees

GROUP BY email

HAVING COUNT(*) > 1;

-- show the full duplicate rows

SELECT * FROM employees

WHERE email IN (

SELECT email FROM employees

GROUP BY email HAVING COUNT(*) > 1

);

Second Highest Salary

-- approach 1: subquery

SELECT MAX(salary) FROM employees

WHERE salary < (SELECT MAX(salary) FROM employees);

-- approach 2: window function (generalises to Nth)

SELECT salary FROM (

SELECT salary, DENSE_RANK() OVER (ORDER BY salary DESC) AS rnk

FROM employees

) ranked WHERE rnk = 2; -- change 2 → N for Nth highest

Cumulative / Running Total

SELECT order_date, amount,

SUM(amount) OVER (ORDER BY order_date ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS running_total

FROM orders;

Month-over-Month Change

WITH monthly AS (

SELECT DATE_TRUNC('month', order_date) AS month,

SUM(amount) AS revenue

FROM orders GROUP BY 1

)

SELECT month, revenue,

LAG(revenue) OVER (ORDER BY month) AS prev_month,

revenue - LAG(revenue) OVER (ORDER BY month) AS change

FROM monthly;

Delete Duplicates — Keep One

-- keep the row with the lowest id, delete the rest

DELETE FROM employees

WHERE id NOT IN (

SELECT MIN(id)

FROM employees

GROUP BY email

);

NULL Handling Patterns

-- replace NULL with a default value

SELECT name, COALESCE(bonus, 0) AS bonus FROM employees;

-- COALESCE returns first non-NULL argument

SELECT COALESCE(mobile, phone, email, 'no contact') AS contact FROM customers;

-- NULLIF — returns NULL if two values are equal (avoid divide-by-zero)

SELECT total / NULLIF(quantity, 0) AS unit_price FROM orders;

-- sort NULLs last explicitly

SELECT * FROM employees ORDER BY bonus DESC NULLS LAST;

Pivot — Rows to Columns with CASE WHEN

-- count employees per department for Q1, Q2, Q3, Q4

SELECT

SUM(CASE WHEN quarter = 1 THEN 1 ELSE 0 END) AS Q1,

SUM(CASE WHEN quarter = 2 THEN 1 ELSE 0 END) AS Q2,

SUM(CASE WHEN quarter = 3 THEN 1 ELSE 0 END) AS Q3,

SUM(CASE WHEN quarter = 4 THEN 1 ELSE 0 END) AS Q4

FROM hires;

Common Pitfalls (Past)

Pitfall (Past)

Rule

NULL comparisons

NULL = NULL → NULL (not TRUE). Always use IS NULL.

NOT IN with NULLs

NOT IN fails silently if subquery contains any NULL. Use NOT EXISTS instead.

SELECT alias in WHERE

Not allowed — WHERE runs before SELECT. Repeat the expression.

GROUP BY order

Non-aggregated SELECT columns must be in GROUP BY.

COUNT(*) vs COUNT(col)

COUNT(*) counts all rows; COUNT(col) skips NULLs.

UPDATE without WHERE

Updates every row. Always double-check the WHERE clause.

DELETE without WHERE

Deletes every row. Always double-check the WHERE clause.

Implicit JOIN

Never use comma join (FROM a, b WHERE a.id=b.x) — use explicit JOIN.

Floating point money

Use DECIMAL/NUMERIC for currency, not FLOAT/REAL.

Full Clause Order Reference

SELECT columns, AGG(col) AS alias, WINDOW_FN() OVER (...) -- 6th: evaluated

FROM table -- 1st

JOIN other ON condition -- 2nd

WHERE row_filter -- 3rd

GROUP BY columns -- 4th

HAVING aggregate_filter -- 5th

ORDER BY columns ASC|DESC -- 7th

LIMIT n OFFSET m; -- 8th
