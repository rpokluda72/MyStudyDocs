1. Co je C#?

C# (čteme „C-Sharp") je moderní, objektově orientovaný programovací jazyk vytvořený společností Microsoft. Spouští se na platformě .NET a poprvé byl vydán v roce 2002. Aktuální verze je C# 13 (2024).

4.1 Aritmetické operátory

Příklad
int a = 10, b = 3;

6.1 Základní syntaxe metody

6.2 Parametry a jejich varianty

Příklad
// Výchozí hodnota parametru
static void Vypis(string zprava, int opakovat = 1)
{
    for (int i = 0; i < opakovat; i++)
        Console.WriteLine(zprava);
}
 
Vypis("Ahoj");         // Ahoj (1×)
Vypis("Test", 3);      // Test (3×)
 
// Pojmenované argumenty
Vypis(opakovat: 2, zprava: "OK");
 
// ref – předání odkazem (může změnit hodnotu)
static void Zdvoj(ref int cislo)
{
    cislo *= 2;
}
int x = 5;
Zdvoj(ref x);
Console.WriteLine(x);   // 10
 
// out – výstupní parametr (musí být přiřazen)
static bool TryDivide(int a, int b, out double vysledek)
{
    if (b == 0) { vysledek = 0; return false; }
    vysledek = (double)a / b;
    return true;
}
 
// params – proměnný počet parametrů
static int Soucet(params int[] cisla)
{
    int s = 0;
    foreach (int c in cisla) s += c;
    return s;
}
Console.WriteLine(Soucet(1, 2, 3, 4, 5));  // 15

6.3 Přetížení metod (Method Overloading)

Příklad
// Stejný název, různé parametry
static int Secti(int a, int b) => a + b;
static double Secti(double a, double b) => a + b;
static string Secti(string a, string b) => a + b;
 
Console.WriteLine(Secti(1, 2));         // 3
Console.WriteLine(Secti(1.5, 2.5));     // 4.0
Console.WriteLine(Secti("A", "B"));     // AB

6.4 Lambda výrazy a expression-body

Příklad
// Expression-body method (=>) – zkrácený zápis
static int Zdvoj(int x) => x * 2;
static bool JeSude(int n) => n % 2 == 0;
 
// Lambda výrazy (anonymní funkce)
Func<int, int> kvadrat = x => x * x;
Func<int, int, int> scitani = (a, b) => a + b;
Action<string> vypis = zprava => Console.WriteLine(zprava);
 
Console.WriteLine(kvadrat(5));        // 25
Console.WriteLine(scitani(3, 4));     // 7
vypis("Lambda!");                     // Lambda!
 
// Predikát
Predicate<int> jeKladne = n => n > 0;

Console.WriteLine(jeKladne(-3));      // False

7.1 Co je OOP?

Objektově orientované programování (OOP) je paradigma, kde program je strukturován jako kolekce objektů. Každý objekt má vlastní stav (data) a chování (metody).

7.2 Třída a objekt

Příklad
// Definice třídy
class Auto
{
    // Pole (field) – privátní
    private string _model;
    private int _rok;
    private double _rychlost;
 
    // Konstruktor
    public Auto(string model, int rok)
    {
        _model = model;
        _rok = rok;
        _rychlost = 0;
    }
 
    // Vlastnost (Property)
    public string Model
    {
        get { return _model; }
        set { _model = value; }
    }
 
    // Auto-implemented property
    public int Rok { get; set; }
 
    // Metoda
    public void Zrychleni(double o)
    {
        _rychlost += o;
        Console.WriteLine($"{_model} jede {_rychlost} km/h");
    }
 
    // Override ToString
    public override string ToString() => $"{_model} ({_rok})";
}
 
// Vytvoření objektu (instance)
Auto mojeAuto = new Auto("Škoda Octavia", 2022);
mojeAuto.Zrychleni(50);      // Škoda Octavia jede 50 km/h
mojeAuto.Zrychleni(30);      // Škoda Octavia jede 80 km/h
Console.WriteLine(mojeAuto); // Škoda Octavia (2022)

8. Dědičnost (Inheritance)

Dědičnost umožňuje třídě (potomek) převzít vlastnosti a metody jiné třídy (rodič). Šetří kód a umožňuje jeho znovupoužití.

┌────────────────────────────┐

│ Třída: Vozidlo │ ← Základní třída (Base class)

│ + znacka: string │

│ + MaxRychlost: int │

│ + Zatroubit() │

└─────────────┬──────────────┘

│ dědí (:)

┌────────┴────────┐

▼ ▼

┌──────────┐ ┌──────────┐

│ Auto │ │ Motor- │ ← Odvozené třídy

│+ model │ │ ka │ (Derived classes)

│+ Jizda() │ │+ Naklonit│

└──────────┘ └──────────┘

Příklad
// Základní třída (Base class / Parent)
class Vozidlo
{
    public string Znacka = "Ford";
 
    public void Zatroubit()
    {
        Console.WriteLine("Tůt, tůt!");
    }
}
 
// Odvozená třída (Derived class / Child)
class Auto : Vozidlo
{
    public string Model = "Mustang";
}
 
// Použití
Auto mojeAuto = new Auto();
mojeAuto.Zatroubit();   // Zděděno z Vozidlo → Tůt, tůt!
Console.WriteLine(mojeAuto.Znacka + " " + mojeAuto.Model);
// Ford Mustang

9. Polymorfismus a abstrakce

9.1 Polymorfismus – virtual a override

Polymorfismus znamená „mnoho forem". Odvozené třídy mohou přepsat chování základní třídy pomocí klíčových slov virtual a override.

Příklad
class Zivocich
{
    public virtual void ZvukZvirete()  // virtual = může být přepsáno
    {
        Console.WriteLine("Zvíře vydává zvuk");
    }
}
 
class Prasatko : Zivocich
{
    public override void ZvukZvirete()  // override = přepisuje rodičovské
    {
        Console.WriteLine("Prase říká: Oink oink");
    }
}
 
class Pes : Zivocich
{
    public override void ZvukZvirete()
    {
        Console.WriteLine("Pes říká: Haf haf");
    }
}
 
// Polymorfní volání – stejný kód, různé chování
Zivocich[] zvirata = { new Zivocich(), new Prasatko(), new Pes() };
foreach (Zivocich z in zvirata)
{
    z.ZvukZvirete();
}
// Zvíře vydává zvuk
// Prase říká: Oink oink
// Pes říká: Haf haf

9.2 Abstraktní třídy (abstract)

Abstraktní třída nemůže být přímo instanciována. Slouží jako šablona pro odvozené třídy.

Příklad
abstract class Tvar
{
    // Abstraktní metoda – nemá tělo, potomci ji MUSÍ implementovat
    public abstract double Obsah();
 
    // Normální metoda – může mít implementaci
    public void Popis()
    {
        Console.WriteLine($"Jsem tvar s obsahem {Obsah():F2}");
    }
}
 
class Kruh : Tvar
{
    private double _polomer;
    public Kruh(double polomer) => _polomer = polomer;
 
    public override double Obsah() => Math.PI * _polomer * _polomer;
}
 
class Obdelnik : Tvar
{
    private double _sirka, _vyska;
    public Obdelnik(double sirka, double vyska)
    {
        _sirka = sirka; _vyska = vyska;
    }
    public override double Obsah() => _sirka * _vyska;
}
 
// Tvar t = new Tvar();  // CHYBA! Abstraktní třídu nelze instanciovat
 
Tvar k = new Kruh(5);
Tvar o = new Obdelnik(4, 6);
k.Popis();   // Jsem tvar s obsahem 78.54
o.Popis();   // Jsem tvar s obsahem 24.00

10. Generické typy (Generics)

Generické typy umožňují psát kód, který pracuje s libovolným typem a přitom zachovává plnou typovou bezpečnost. Místo konkrétního typu se použije typový parametr T.

10.1 Generická třída

Příklad
// Generická třída – T je typový parametr
class Zasobnik<T>
{
    private List<T> _prvky = new List<T>();
 
    public void Vloz(T prvek) => _prvky.Add(prvek);
    public T Vyjmi()
    {
        T posledni = _prvky[_prvky.Count - 1];
        _prvky.RemoveAt(_prvky.Count - 1);
        return posledni;
    }
    public int Pocet => _prvky.Count;
}
 
// Použití s různými typy
var intZasobnik = new Zasobnik<int>();
intZasobnik.Vloz(1);
intZasobnik.Vloz(2);
Console.WriteLine(intZasobnik.Vyjmi());  // 2
 
var strZasobnik = new Zasobnik<string>();
strZasobnik.Vloz("Ahoj");
strZasobnik.Vloz("Svět");
Console.WriteLine(strZasobnik.Vyjmi()); // Svět

10.2 Generická metoda

Příklad
// Generická metoda – typový parametr T je součástí metody

{
    Array.Reverse(pole);
    return pole;
}
 
// Kompilátor odvozuje T z argumentu

    Console.WriteLine($"Hodnota: {hodnota} (typ: {typeof(T).Name})");

Příklad
// where T : IComparable<T> – T musí podporovat porovnávání

    => a.CompareTo(b) >= 0 ? a : b;
 
Console.WriteLine(Maximum(3, 7));            // 7
Console.WriteLine(Maximum("apple", "banana")); // banana
 
// Kombinace omezení

    => new T();

13. LINQ – Language Integrated Query

LINQ (Language Integrated Query) je sada technologií pro dotazování nad datovými zdroji přímo v jazyce C#. Dotazy jsou typově bezpečné a plně integrovány do jazyka.

LINQ umožňuje jednotný způsob dotazování nad poli, kolekcemi, XML, databází (EF Core) a dalšími zdroji dat pomocí stejné syntaxe.

13.1 Dotazová syntaxe (Query Syntax)

Příklad
int[] skore = { 97, 45, 81, 60, 75, 92 };
 
// Dotaz: vyber skóre > 70, seřaď sestupně
IEnumerable<int> dotaz =
    from s in skore
    where s > 70
    orderby s descending
    select s;
 
foreach (var s in dotaz)
    Console.Write(s + " ");
// Výstup: 97 92 81 75

Příklad

14. Asynchronní programování (async/await)

Asynchronní programování umožňuje provádět dlouhotrvající operace (I/O, síť, databáze) bez blokování hlavního vlákna. Klíčová slova async a await zjednodušují práci s úlohami.

14.1 Synchronní vs asynchronní

  SYNCHRONNÍ (blokující):
  ──────────────────────────────────────────────────►
  │ Task 1: DB query (500ms) │ Task 2: API call (300ms) │ Task 3 │
  Celkem: ~800ms
 
  ASYNCHRONNÍ (neblokující):
  ──────────────────────────────────────────────────►
  │ Task 1: DB query  ─────────────────────────► │
  │ Task 2: API call  ────────────────►           │
  │ Task 3 (spuštěn ihned)  ──────────►           │
  Celkem: ~500ms (nejdelší task)

14.2 async / await – základy

Příklad
using System.Threading.Tasks;
 
// async metoda vrací Task nebo Task<T>
static async Task<string> StahniDataAsync(string url)
{
    using var client = new System.Net.Http.HttpClient();
    string obsah = await client.GetStringAsync(url);  // neblokuje vlákno
    return obsah;
}
 
// Volání async metody
static async Task Main(string[] args)
{
    Console.WriteLine("Stahuji...");
    string data = await StahniDataAsync("https://api.example.com/data");
    Console.WriteLine($"Staženo: {data.Length} znaků");
 
    // await Task.Delay – neblokující čekání
    await Task.Delay(1000);  // čeká 1 sekundu bez blokování vlákna
    Console.WriteLine("Hotovo!");
}

14.3 Paralelní úlohy – Task.WhenAll a Task.WhenAny

Příklad
static async Task<int> ZiskejDataAsync(int id)
{
    await Task.Delay(id * 100);  // simulace různé doby zpracování
    return id * id;
}
 
static async Task Main()
{
    // Spustí všechny úlohy paralelně, čeká na dokončení VŠECH
    Task<int>[] ukoly = {
        ZiskejDataAsync(1),
        ZiskejDataAsync(2),
        ZiskejDataAsync(3),
    };
    int[] vysledky = await Task.WhenAll(ukoly);
    Console.WriteLine(string.Join(", ", vysledky));  // 1, 4, 9
 
    // Čeká na PRVNÍ dokončenou úlohu
    Task<int> prvni = await Task.WhenAny(
        ZiskejDataAsync(3),
        ZiskejDataAsync(1),
        ZiskejDataAsync(2)
    );
    Console.WriteLine($"Nejrychlejší výsledek: {await prvni}");  // 1
}

Příklad
static async Task DlouhaOperaceAsync(CancellationToken token)
{
    for (int i = 0; i < 10; i++)
    {
        token.ThrowIfCancellationRequested();  // vyhodí výjimku při zrušení

    catch (OperationCanceledException)
    {
        Console.WriteLine("Operace byla zrušena!");
    }
}

Příklad
using System.IO;
 
// Zápis přes StreamWriter
using (StreamWriter sw = new StreamWriter("data.txt"))
{
    sw.WriteLine("Řádek 1");
    sw.WriteLine("Řádek 2");
}  // using zajistí automatické uzavření (Dispose)
 
// Čtení přes StreamReader
using (StreamReader sr = new StreamReader("data.txt"))
{
    string? radek;
    while ((radek = sr.ReadLine()) != null)
    {
        Console.WriteLine(radek);
    }
}
 
// Asynchronní I/O (doporučeno)
await File.WriteAllTextAsync("async.txt", "Asynchronní zápis");
string data = await File.ReadAllTextAsync("async.txt");
